If the automatic fix doesn't work, run this in your browser console:

// Force set authentication state
localStorage.setItem('selectedRole', 'Director');

// Check if using mock auth
const useMock = import.meta?.env?.VITE_USE_MOCK_AUTH === 'true';
console.log('Using mock auth:', useMock);

// If using real auth, ensure tokens are present
if (!useMock) {
    const token = sessionStorage.getItem('accessToken') || localStorage.getItem('accessToken');
    const user = sessionStorage.getItem('user') || localStorage.getItem('user');
    console.log('Token present:', !!token);
    console.log('User present:', !!user);
    
    if (!token || !user) {
        console.log('❌ Missing real auth data - need to login properly');
    }
}

// Force reload the app
window.location.reload();
